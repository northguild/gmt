# PROTOTYPE, throwaway. DOX-G2 live preview: proxies the Dox dev server and injects
# __proto/proto.css + proto.js into every HTML page. Nothing in the repo changes.
# "/" serves the live northguild.dev home with slice 5's companion CSS, so a click on its
# GMT link runs the cross-document view transition into /gmt/ on one origin.
import http.server, urllib.request, os, sys
UP = "http://localhost:4321"
NG = "https://northguild.dev"
HERE = os.path.dirname(os.path.abspath(__file__))
INJECT = b'<script>addEventListener("pagereveal",e=>{window.__pVT=e.viewTransition?"transition":"none"})</script><script>if(/[?&]slow/.test(location.search))document.documentElement.classList.add("p-slow")</script><link rel="stylesheet" href="/__proto/proto.css"><script type="module" src="/__proto/proto.js"></script>'
NG_CSS = b"""<style>
@view-transition { navigation: auto; }
.site-header { view-transition-name: ng-header; block-size: 3.5rem; box-sizing: border-box; }
.site-header .ng-mark { view-transition-name: ng-mark; }
.site-header .ng-wordmark { view-transition-name: ng-wordmark; }
::view-transition-group(ng-mark), ::view-transition-group(ng-wordmark),
::view-transition-new(ng-mark), ::view-transition-new(ng-wordmark) { animation: none; }
::view-transition-old(ng-mark), ::view-transition-old(ng-wordmark) { display: none; }
::view-transition-old(ng-header), ::view-transition-new(ng-header) { animation-duration: 240ms; animation-timing-function: ease-out; }
::view-transition-old(root), ::view-transition-new(root) { animation-duration: 600ms; animation-timing-function: cubic-bezier(.25,.6,.2,1); }
@media (prefers-reduced-motion: reduce) { @view-transition { navigation: none; } }
</style>"""
class H(http.server.BaseHTTPRequestHandler):
    def log_message(self, *a): pass
    def send(self, status, ct, body, extra=()):
        self.send_response(status); self.send_header("Content-Type", ct); self.send_header("Cache-Control", "no-store")
        for k, v in extra: self.send_header(k, v)
        self.send_header("Content-Length", str(len(body))); self.end_headers(); self.wfile.write(body)
    def do_GET(self):
        path = self.path.split("?")[0]
        if path.startswith("/__proto/"):
            p = os.path.join(HERE, path.lstrip("/"))
            if not os.path.isfile(p): self.send_error(404); return
            ct = {"css": "text/css", "js": "text/javascript", "json": "application/json"}[p.rsplit(".", 1)[1]]
            self.send(200, ct, open(p, "rb").read()); return
        if path == "/" or path.startswith("/_astro/") or path == "/favicon.svg":
            r = urllib.request.urlopen(urllib.request.Request(NG + self.path, headers={"User-Agent": "gmt-preview"}))
            body, ct = r.read(), r.headers.get("Content-Type", "")
            if path == "/":
                body = body.replace(b"</head>", NG_CSS + b"</head>", 1)
                body = body.replace(b"</svg>NorthGuild</div>", b'</svg><span class="ng-wordmark">NorthGuild</span></div>', 1)
                body = body.replace(b'href="https://northguild.dev/gmt/"', b'href="/gmt/"')
            self.send(200, ct, body); return
        hdrs = {k: v for k, v in self.headers.items() if k.lower() not in ("host", "accept-encoding", "connection", "upgrade")}
        req = urllib.request.Request(UP + self.path, headers=hdrs)
        try:
            r = urllib.request.urlopen(req)
        except urllib.error.HTTPError as e:
            r = e
        body = r.read()
        ct = r.headers.get("Content-Type", "")
        if "text/html" in ct and b"</head>" in body:
            body = body.replace(b"</head>", INJECT + b"</head>", 1)
            # The mark swap runs before the header's first paint, so the view transition snapshots it.
            body = body.replace(b"</header>", b'<script src="/__proto/header.js"></script></header>', 1)
        self.send_response(r.status)
        for k, v in r.headers.items():
            if k.lower() in ("content-length", "transfer-encoding", "connection", "content-encoding"): continue
            self.send_header(k, v)
        self.send_header("Content-Length", str(len(body))); self.end_headers(); self.wfile.write(body)
http.server.ThreadingHTTPServer(("127.0.0.1", int(sys.argv[1]) if len(sys.argv) > 1 else 4748), H).serve_forever()
